cmake_minimum_required(VERSION 3.8)
project(robot_head)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/head_pitch_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_head_policy test/test_head_policy.py)
  ament_add_pytest_test(test_head_interfaces test/test_head_interfaces.py)
  ament_add_pytest_test(test_head_gazebo test/test_head_gazebo.py NOCAPTURE)
  set_tests_properties(test_head_gazebo PROPERTIES LABELS "pytest;gazebo" TIMEOUT 0)
endif()
ament_package()
